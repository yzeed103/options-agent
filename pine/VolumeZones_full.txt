// ═══════════════ Colour helper ═══════════════
// The colour pickers already carry their own transparency. Older code used color.new(c, X), which REPLACES that
// transparency, so the picker's opacity slider had no effect. fade() multiplies the two opacities instead:
// picker 50% + setting 20% -> 60% transparent. Either control can fade a colour and neither overrides the other.
fade(color c, float extraT) => color.new(c, 100 - (100 - color.t(c)) * (100 - math.max(0, math.min(100, extraT))) / 100)

// Keep the heat-map thresholds in order even if the user sets them crossed: hidden < medium <= high.
heatMidE  = math.max(heatMid, heatMin)
heatHighE = math.max(heatHigh, heatMidE)

// ═══════════════ Volume / trend ═══════════════
atrV  = ta.atr(14)
volMa = ta.sma(volume, volMaLen)
vr    = volMa > 0 ? volume / volMa : 1.0
weak  = vr < weakRatio
strongVol = vr >= breakRatio
[diP, diM, adxV] = ta.dmi(adxLen, adxLen)
emaF = ta.ema(close, 20)
emaS = ta.ema(close, 50)
trendDir = emaF > emaS and diP > diM ? 1 : emaF < emaS and diM > diP ? -1 : 0

// ═══════════════ Rolling volume profile (bars 1..lookback, past only) ═══════════════
var float[] rowVol = array.new_float(rows, 0.0)
var float[] rowBuy = array.new_float(rows, 0.0)
var float[] hvnLo  = array.new_float()
var float[] hvnHi  = array.new_float()
pHi  = ta.highest(high, lookback)[1]
pLo  = ta.lowest(low, lookback)[1]
step = (pHi - pLo) / rows
float poc = na
float vah = na
float val = na
ready = bar_index > lookback and step > 0

if ready
    array.fill(rowVol, 0.0)
    array.fill(rowBuy, 0.0)
    for i = 1 to lookback
        bh = high[i]
        bl = low[i]
        bv = nz(volume[i])
        isUp = close[i] >= open[i]
        rng = math.max(bh - bl, syminfo.mintick)
        r0 = math.max(0, math.min(rows - 1, int((bl - pLo) / step)))
        r1 = math.max(0, math.min(rows - 1, int((bh - pLo) / step)))
        for r = r0 to r1
            rLo = pLo + r * step
            rHi = rLo + step
            ov = math.max(0.0, math.min(bh, rHi) - math.max(bl, rLo))
            add = bh == bl ? bv : bv * ov / rng
            array.set(rowVol, r, array.get(rowVol, r) + add)
            if isUp
                array.set(rowBuy, r, array.get(rowBuy, r) + add)
    total = array.sum(rowVol)
    pocI = array.indexof(rowVol, array.max(rowVol))
    loI = pocI
    hiI = pocI
    acc = array.get(rowVol, pocI)
    while acc < vaPct * total and (loI > 0 or hiI < rows - 1)
        upV1 = hiI < rows - 1 ? array.get(rowVol, hiI + 1) : -1.0
        dnV1 = loI > 0 ? array.get(rowVol, loI - 1) : -1.0
        if upV1 >= dnV1
            hiI += 1
            acc += upV1
        else
            loI -= 1
            acc += dnV1
    poc := pLo + (pocI + 0.5) * step
    vah := pLo + (hiI + 1) * step
    val := pLo + loI * step
    array.clear(hvnLo)
    array.clear(hvnHi)
    mean = total / rows
    inRun = false
    for r = 0 to rows - 1
        isH = array.get(rowVol, r) > hvnMult * mean
        if isH and not inRun
            array.push(hvnLo, pLo + r * step)
            array.push(hvnHi, pLo + (r + 1) * step)
            inRun := true
        else if isH and inRun
            array.set(hvnHi, array.size(hvnHi) - 1, pLo + (r + 1) * step)
        else
            inRun := false

nZones = array.size(hvnLo)

// ═══════════════ Nearest targets (HVN edges, POC, VA edges) ═══════════════
float tgtUp = na
float tgtDn = na
cands = array.new_float()
if nZones > 0
    for k = 0 to nZones - 1
        array.push(cands, array.get(hvnLo, k))
        array.push(cands, array.get(hvnHi, k))
if not na(poc)
    array.push(cands, poc)
    array.push(cands, vah)
    array.push(cands, val)
if array.size(cands) > 0 and atrV > 0
    for k = 0 to array.size(cands) - 1
        x = array.get(cands, k)
        dist = math.abs(x - close) / atrV
        if dist >= minTgtAtr and dist <= maxTgtAtr
            if x > close and (na(tgtUp) or x < tgtUp)
                tgtUp := x
            if x < close and (na(tgtDn) or x > tgtDn)
                tgtDn := x
inHvn = false
if nZones > 0
    for k = 0 to nZones - 1
        if close >= array.get(hvnLo, k) and close <= array.get(hvnHi, k)
            inHvn := true

// ═══════════════ Weak-volume pool → break → HUNT ═══════════════
var int poolStart = na
winRng  = ta.highest(high, weakBars) - ta.lowest(low, weakBars)
weakCnt = math.sum(weak ? 1 : 0, weakBars)
if na(poolStart) and weakCnt >= weakBars - 1 and winRng <= poolMaxAtr * atrV
    poolStart := bar_index - weakBars + 1

int brk = 0
int hunt = 0
float poolH = na
float poolL = na
float huntTgt = na
if not na(poolStart)
    len = math.min(bar_index - poolStart, 400)
    if len >= 1
        float ph = high[1]
        float pl = low[1]
        for i = 1 to len
            ph := math.max(ph, high[i])
            pl := math.min(pl, low[i])
        poolH := ph
        poolL := pl
    if not na(poolH) and close > poolH and strongVol
        brk := 1
    else if not na(poolL) and close < poolL and strongVol
        brk := -1
    if brk != 0
        t = brk > 0 ? tgtUp : tgtDn
        emaOk = brk > 0 ? emaF > emaS : emaF < emaS
        if not na(t) and adxV >= adxStrong and emaOk
            hunt := brk
            huntTgt := t
        poolStart := na
    else if not na(poolH) and ((poolH - poolL) > poolMaxAtr * atrV or strongVol)
        poolStart := na   // pool widened, or strong volume inside it without a break: pool is over

// ═══════════════ Rejection at an HVN edge ═══════════════
int rej = 0
rngB = high - low
if rngB > 0 and nZones > 0
    uw = (high - math.max(open, close)) / rngB
    lw = (math.min(open, close) - low) / rngB
    for k = 0 to nZones - 1
        zLo = array.get(hvnLo, k)
        zHi = array.get(hvnHi, k)
        if low < zLo and high >= zLo and close < zLo and close < open and uw >= wickPct
            rej := -1
        if low <= zHi and high > zHi and close > zHi and close > open and lw >= wickPct
            rej := 1
